find_package(doctest QUIET)
if(NOT TARGET doctest::doctest)
    find_path(DOCTEST_INCLUDE_DIR NAMES doctest/doctest.h)
    if(DOCTEST_INCLUDE_DIR)
        add_library(doctest::doctest INTERFACE IMPORTED)
        set_target_properties(doctest::doctest PROPERTIES
            INTERFACE_INCLUDE_DIRECTORIES "${DOCTEST_INCLUDE_DIR}")
    endif()
endif()

if(TARGET doctest::doctest)
    add_executable(uni_tests tests.cpp)
    target_link_libraries(uni_tests PRIVATE unicore doctest::doctest)
    target_compile_features(uni_tests PRIVATE cxx_std_23)
    uni_enable_warnings(uni_tests)
    set_target_properties(uni_tests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
    add_test(NAME uni_tests COMMAND uni_tests)

    if(TARGET unitui)
        add_executable(uni_tui_tests tui_tests.cpp)
        target_link_libraries(uni_tui_tests PRIVATE unitui doctest::doctest)
        target_compile_features(uni_tui_tests PRIVATE cxx_std_23)
        uni_enable_warnings(uni_tui_tests)
        set_target_properties(uni_tui_tests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
        add_test(NAME uni_tui_tests COMMAND uni_tui_tests)
    endif()
else()
    message(STATUS "doctest not found; unit tests disabled")
endif()
